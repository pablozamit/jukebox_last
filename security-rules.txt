rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Regla global para Admins (UID en la colección 'admins').
    // Los admins (panel /admin y bridge) pueden leer/escribir cualquier documento.
    match /{document=**} {
      allow read, write: if request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // ============ 1. songs — Cola activa ============
    // Cualquiera puede ver la cola. Votar = crear (1 voto) o incrementar/decrementar en 1.
    match /songs/{songId} {
      allow read: if true;

      // Proponer una canción nueva: el documento nace con exactamente 1 voto.
      allow create: if request.auth != null
                    && request.resource.data.votes == 1
                    && request.resource.data.keys().hasAll(['title', 'votes'])
                    && request.resource.data.keys().hasOnly(['title', 'votes', 'firstVotedAt', 'proposerName', 'lastVoterName']);

      // Votar (+1) o retirar voto (-1), y nada más.
      allow update: if request.auth != null
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes', 'firstVotedAt', 'proposerName', 'lastVoterName'])
                    && (request.resource.data.votes == resource.data.votes + 1
                        || request.resource.data.votes == resource.data.votes - 1);

      // Retirar el último voto borra la canción de la cola.
      allow delete: if request.auth != null
                    && resource.data.votes <= 1;
    }

    // ============ 2. catalog — Solo lectura (escribe el bridge / admin) ============
    match /catalog/full_list {
      allow read: if true;
      allow write: if false;
    }

    // ============ 3. state — Solo lectura (escribe el bridge con Admin SDK) ============
    match /state/{document=**} {
      allow read: if true;
      allow write: if false;
    }

    // ============ 4. users — Cada usuario solo lee/escribe SU propio documento ============
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;

      // Creación inicial del documento (login anónimo): solo arrays vacíos.
      allow create: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.keys().hasOnly(['proposals', 'votes']);

      // Actualización: solo campos controlados (votos, registro, DJ name, puntos, favoritos).
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['proposals', 'votes', 'freeVotes', 'isRegistered', 'email', 'djName', 'points', 'favorites']);
    }

    // ============ 5. commands — Solo admins (regla global) ============
    match /commands/{commandId} {
      allow read: if false;
      allow write: if false;
    }

    // ============ 6. suggestions — Cualquiera puede sugerir; solo admins leen/borran ============
    match /suggestions/{docId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }

    // ============ 7. statistics — Votos, tiempos y eventos públicos ============
    match /statistics/{docId} {
      allow read: if true;

      // Eventos de voto (addDoc con auto-ID): cualquiera logueado puede crear SOLO vote_events.
      // (No se permiten crear otros docs en statistics: evitaría docs basura arbitrarios.)
      allow create: if request.auth != null
                    && request.resource.data.kind == 'vote_event';

      // Contadores de votos/tiempo: solo los documentos canónicos de la app.
      allow update: if request.auth != null
                    && docId.matches('^(votes|time)_(hoy|semana|mes|total)$');

      // Limpieza automática de vote_events viejos (>2h): solo esos documentos y solo borrar.
      allow delete: if request.auth != null
                    && resource.data.kind == 'vote_event'
                    && request.time.toMillis() - resource.data.ts > (2 * 60 * 60 * 1000);
    }

    // ============ 8. admins — Solo lectura del propio documento ============
    match /admins/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if false;
    }

    // ============ 9. leaderboard — Solo lectura (escribe el bridge) ============
    match /leaderboard/{docId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ 10. design_feedback — Encuesta de diseño ============
    match /design_feedback/{docId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }
  }
}
