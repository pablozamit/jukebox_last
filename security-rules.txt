rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Regla global para Admins (UID en la colección 'admins').
    // Los admins (panel /admin y bridge) pueden leer/escribir cualquier documento.
    match /{document=**} {
      allow read, write: if request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // ============ 1. songs — Cola activa ============
    // Cualquiera puede ver la cola. Votar = crear (1 voto) o incrementar/decrementar en 1.
    match /songs/{songId} {
      allow read: if true;

      // Proponer una canción nueva: el documento nace con exactamente 1 voto.
      allow create: if request.auth != null
                    && request.resource.data.votes == 1
                    && request.resource.data.keys().hasAll(['title', 'votes'])
                    && request.resource.data.keys().hasOnly(['title', 'votes', 'firstVotedAt', 'proposerName', 'lastVoterName']);

      // Votar (+1) o retirar voto (-1), y nada más.
      allow update: if request.auth != null
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes', 'firstVotedAt', 'proposerName', 'lastVoterName'])
                    && (request.resource.data.votes == resource.data.votes + 1
                        || request.resource.data.votes == resource.data.votes - 1);

      // Retirar el último voto borra la canción de la cola.
      allow delete: if request.auth != null
                    && resource.data.votes <= 1;
    }

    // ============ 2. catalog — Solo lectura (escribe el bridge / admin) ============
    match /catalog/full_list {
      allow read: if true;
      allow write: if false;
    }

    // ============ 3. state — Solo lectura (escribe el bridge con Admin SDK) ============
    match /state/{document=**} {
      allow read: if true;
      allow write: if false;
    }

    // ============ 4. users — Cada usuario solo lee/escribe SU propio documento ============
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;

      // Creación inicial del documento (login anónimo o registro inmediato).
      // Siempre deben existir los dos arrays; el resto son campos de perfil controlados.
      allow create: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.keys().hasAll(['proposals', 'votes'])
                    && request.resource.data.keys().hasOnly(['proposals', 'votes', 'isRegistered', 'email', 'djName'])
                    && (!request.resource.data.keys().hasAny(['isRegistered', 'email', 'djName'])
                        || (request.resource.data.isRegistered == true
                            && request.resource.data.email == request.auth.token.email));

      // El cliente solo puede editar sus votos/propuestas, favoritos y nombre DJ.
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['proposals', 'votes', 'djName', 'favorites']);

      // Completar el registro después de enlazar la cuenta anónima con email/password.
      // El email guardado debe coincidir con el email del token de Firebase Auth.
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['proposals', 'votes', 'isRegistered', 'email', 'djName'])
                    && request.resource.data.isRegistered == true
                    && request.resource.data.email == request.auth.token.email;

      // La trivia puede conceder exactamente +1 propuesta y +2 votos una vez cada
      // 30 minutos. Los puntos y demás campos de servidor no son editables por el cliente.
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['freeProposals', 'freeVotes', 'lastTriviaAt'])
                    && request.time.toMillis() - resource.data.get('lastTriviaAt', 0) >= 30 * 60 * 1000
                    && request.resource.data.freeProposals == resource.data.get('freeProposals', 0) + 1
                    && request.resource.data.freeVotes == resource.data.get('freeVotes', 0) + 2
                    && request.resource.data.lastTriviaAt >= request.time.toMillis() - 60 * 1000;

      // Perder una ronda solo actualiza la marca de tiempo, con el mismo cooldown.
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lastTriviaAt'])
                    && request.time.toMillis() - resource.data.get('lastTriviaAt', 0) >= 30 * 60 * 1000
                    && request.resource.data.lastTriviaAt >= request.time.toMillis() - 60 * 1000;

      // Canjear 50 puntos por un voto gratis: operación legítima y acotada.
      // No permite fabricar puntos ni votos; exige saldo suficiente y un cambio exacto.
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['points', 'freeVotes'])
                    && resource.data.get('points', 0) >= 50
                    && request.resource.data.points == resource.data.get('points', 0) - 50
                    && request.resource.data.freeVotes == resource.data.get('freeVotes', 0) + 1;
    }

    // ============ 5. commands — Solo admins (regla global) ============
    match /commands/{commandId} {
      allow read: if false;
      allow write: if false;
    }

    // ============ 6. suggestions — Cualquiera puede sugerir; solo admins leen/borran ============
    match /suggestions/{docId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }    // ============ 7. statistics — Contadores protegidos ============
    match /statistics/{docId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ 8. voteEvents — Evento anidado bajo la canción ============
    // La ruta aporta songId de forma estática a las reglas. El evento solo puede
    // crearse en la misma transacción que actualiza el usuario y la canción.
    match /songs/{songId}/voteEvents/{eventId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.resource.data.keys().hasOnly(['kind', 'userId', 'songId', 'title', 'type', 'voterName', 'ts'])
                    && request.resource.data.keys().hasAll(['kind', 'userId', 'songId', 'title', 'type', 'voterName', 'ts'])
                    && request.resource.data.kind == 'vote_event'
                    && request.resource.data.userId == request.auth.uid
                    && request.resource.data.songId == songId
                    && (request.resource.data.type == 'vote' || request.resource.data.type == 'proposal')
                    && request.resource.data.title is string
                    && request.resource.data.title.size() > 0
                    && request.resource.data.title.size() <= 240
                    && request.resource.data.voterName is string
                    && request.resource.data.voterName.size() > 0
                    && request.resource.data.voterName.size() <= 40
                    && request.resource.data.ts is int
                    && request.resource.data.ts >= request.time.toMillis() - (10 * 60 * 1000)
                    && request.resource.data.ts <= request.time.toMillis() + 60000
                    && (
                      (request.resource.data.type == 'proposal'
                        && !exists(/databases/$(database)/documents/songs/$(songId))
                        && getAfter(/databases/$(database)/documents/songs/$(songId)).data.votes == 1
                        && getAfter(/databases/$(database)/documents/songs/$(songId)).data.title == request.resource.data.title
                        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data.proposals.hasAll([songId]))
                      ||
                      (request.resource.data.type == 'vote'
                        && exists(/databases/$(database)/documents/songs/$(songId))
                        && getAfter(/databases/$(database)/documents/songs/$(songId)).data.votes
                           == get(/databases/$(database)/documents/songs/$(songId)).data.votes + 1
                        && getAfter(/databases/$(database)/documents/songs/$(songId)).data.title == request.resource.data.title
                        && getAfter(/databases/$(database)/documents/users/$(request.auth.uid)).data.votes.hasAll([songId]))
                    );

      allow update, delete: if false;
    }

    // ============ 9. admins — Solo lectura del propio documento ============


    // ============ 9. admins — Solo lectura del propio documento ============
    match /admins/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if false;
    }

    // ============ 10. leaderboard — Solo lectura (escribe el bridge) ============
    match /leaderboard/{docId} {
      allow read: if true;
      allow write: if false;
    }

    // ============ 11. design_feedback — Encuesta de diseño ============
    match /design_feedback/{docId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }
  }
}
