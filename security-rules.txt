rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    
    // Regla Global para Admins
    // Otorga acceso total si el UID del usuario está en la colección 'admins'
    match /{document=**} {
      allow read, write: if request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    // 1. Colección de canciones (songs) - Cola Activa
    match /songs/{songId} {
      // Todos pueden leer las canciones para ver la cola y los votos
      allow read: if true;
      
      // Permitir la CREACIÓN si:
      // a) El usuario está logueado.
      // b) El valor inicial de votos es 1.
      allow create: if request.auth != null
                    && request.resource.data.votes == 1;

      // Permitir la ACTUALIZACIÓN de votos SOLO si:
      // a) El usuario está logueado.
      // b) Solo se está modificando 'votes' y 'firstVotedAt'.
      // c) El incremento es +1 (votar) o -1 (retirar voto).
      allow update: if request.auth != null 
                    && (request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes', 'firstVotedAt', 'lastVoterName', 'proposerName']))
                    && (request.resource.data.votes == resource.data.votes + 1
                        || request.resource.data.votes == resource.data.votes - 1);
                    
      // Permitir BORRADO si:
      // a) El usuario está logueado.
      // b) La canción tiene 1 voto o menos (era el último voto al retirarlo).
      // Los admins pueden borrar cualquier canción vía la regla global.
      allow delete: if request.auth != null
                    && resource.data.votes <= 1;
    }

    // 2. Catálogo completo (Estructura estática)
    match /catalog/full_list {
      allow read: if true;
      allow write: if false; // Solo admins (regla global)
    }

    // 3. Estado de reproducción (state)
    match /state/{document=**} {
      allow read: if true;
      allow write: if false; 
    }

    // 4. Usuarios y tokens (users)
    // El usuario puede leer su propio documento completo.
    // Solo puede escribir campos seguros (no puntos, logros ni historial).
    // El bridge (Admin SDK) bypassa estas reglas para escribir puntos, etc.
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.keys().hasOnly(['proposals', 'votes', 'djName']);
      allow update: if request.auth != null && request.auth.uid == userId
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['proposals', 'votes', 'freeVotes', 'isRegistered', 'email', 'djName', 'points']);
    }

    // 5. Comandos (commands)
    match /commands/{commandId} {
      allow read: if false;
      allow write: if false; // Solo admins (regla global)
    }

    // 6. Sugerencias (suggestions)
    match /suggestions/{docId} {
      allow create: if request.auth != null;
      allow read, update, delete: if false;
    }

    // 7. Estadísticas (statistics)
    match /statistics/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    // 8. Colección de administradores (admins)
    // Un usuario puede leer SI PROPIO documento para saber si es admin.
    // La escritura solo se hace via Admin SDK (Python) o por la regla global.
    match /admins/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if false;
    }

    // 9. Leaderboard semanal de DJs
    // Todos pueden leer (para ver el ranking). Solo el bridge escribe.
    match /leaderboard/{docId} {
      allow read: if true;
      allow write: if false;
    }
  }
}
